```typescript
// NFR-DATA-01: 필드별 마스킹 정책
export const MASKING_POLICY = {
  phone: (value: string) => {
    // 010-****-1234
    return value.replace(/(\d{3})-?(\d{4})-?(\d{4})/, '$1-****-$3')
  },
  email: (value: string) => {
    // t***@example.com
    const [local, domain] = value.split('@')
    return `${local[0]}***@${domain}`
  },
  accountNumber: (value: string) => {
    // ****-****-****-1234
    return value.slice(-4).padStart(value.length, '*')
  },
  name: (value: string) => {
    // 홍*동
    if (value.length <= 2) return value[0] + '*'
    return value[0] + '*'.repeat(value.length - 2) + value.slice(-1)
  },
}
```

```typescript
// NFR-DATA-02: 대량 다운로드 통제
export const DOWNLOAD_LIMITS = {
  maxRowsPerDownload: 1000,     // 최대 행수 제한
  dailyQuota: 5000,             // 일일 쿼터
  rateLimitPerMinute: 3,        // 분당 다운로드 횟수
}

// 다운로드 가능 여부 체크
const useDownloadQuota = () => {
  const [todayDownloads, setTodayDownloads] = useState(0)
  const [lastDownloadAt, setLastDownloadAt] = useState<Date | null>(null)

  const canDownload = (rowCount: number) => {
    // 행수 제한 체크
    if (rowCount > DOWNLOAD_LIMITS.maxRowsPerDownload) {
      return { allowed: false, reason: `최대 ${DOWNLOAD_LIMITS.maxRowsPerDownload}행까지 다운로드 가능합니다.` }
    }

    // 일일 쿼터 체크
    if (todayDownloads + rowCount > DOWNLOAD_LIMITS.dailyQuota) {
      return { allowed: false, reason: '일일 다운로드 한도를 초과했습니다.' }
    }

    // 레이트 리밋 체크
    if (lastDownloadAt && Date.now() - lastDownloadAt.getTime() < 20000) {
      return { allowed: false, reason: '잠시 후 다시 시도해주세요.' }
    }

    return { allowed: true }
  }

  return { canDownload, todayDownloads }
}
```

```typescript
// NFR-SEC-AUT-02: 로그인 레이트리밋
export const LOGIN_POLICY = {
  maxFailedAttempts: 5,         // 5회 실패 시 잠금
  lockoutDurationMinutes: 30,   // 잠금 시간
  cooldownAfterFailure: 3000,   // 실패 후 대기 시간 (ms)
}

const useLogin = () => {
  const [failedAttempts, setFailedAttempts] = useState(0)
  const [isLocked, setIsLocked] = useState(false)
  const [lockUntil, setLockUntil] = useState<Date | null>(null)

  const login = async (credentials: LoginForm) => {
    if (isLocked) {
      throw new Error('계정이 잠겼습니다. 잠시 후 다시 시도해주세요.')
    }

    try {
      const result = await authService.login(credentials)
      setFailedAttempts(0)
      return result
    } catch (error) {
      const newAttempts = failedAttempts + 1
      setFailedAttempts(newAttempts)

      if (newAttempts >= LOGIN_POLICY.maxFailedAttempts) {
        setIsLocked(true)
        setLockUntil(new Date(Date.now() + LOGIN_POLICY.lockoutDurationMinutes * 60 * 1000))
      }

      throw error
    }
  }

  return { login, failedAttempts, isLocked, lockUntil }
}
```

#### 2) 세션 만료 처리
```typescript
// NFR-SEC-AUT-02: 세션 만료 정책
export const SESSION_POLICY = {
  maxIdleMinutes: 30,           // 비활성 30분 후 만료
  absoluteMaxMinutes: 480,      // 절대 최대 8시간
  warningBeforeExpireMinutes: 5, // 만료 5분 전 경고
}

const useSessionTimeout = () => {
  const [showWarning, setShowWarning] = useState(false)

  useEffect(() => {
    let idleTimer: NodeJS.Timeout
    let warningTimer: NodeJS.Timeout

    const resetTimer = () => {
      clearTimeout(idleTimer)
      clearTimeout(warningTimer)

      // 경고 타이머
      warningTimer = setTimeout(() => {
        setShowWarning(true)
      }, (SESSION_POLICY.maxIdleMinutes - SESSION_POLICY.warningBeforeExpireMinutes) * 60 * 1000)

      // 만료 타이머
      idleTimer = setTimeout(() => {
        logout()
      }, SESSION_POLICY.maxIdleMinutes * 60 * 1000)
    }

    // 사용자 활동 감지
    const events = ['mousedown', 'keydown', 'scroll', 'touchstart']
    events.forEach(event => window.addEventListener(event, resetTimer))

    resetTimer()

    return () => {
      clearTimeout(idleTimer)
      clearTimeout(warningTimer)
      events.forEach(event => window.removeEventListener(event, resetTimer))
    }
  }, [])

  return { showWarning, extendSession: () => setShowWarning(false) }
}
```
