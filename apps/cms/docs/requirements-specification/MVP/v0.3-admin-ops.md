### 구현 항목

#### 1) 참여자 조회 페이지 (`/admin/users/participants`)
```typescript
// FR-F00: 참여자 조회 (프로그램별)
interface ParticipantListFilters {
  programId?: string            // 프로그램 필터
  role?: 'INDIVIDUAL' | 'SCHOOL' // 역할 필터
  status?: string               // 신청 상태 필터
  keyword?: string              // 검색어 (이름/이메일)
}

// 다운로드 권한 체크
const canDownload = (user: User, programId: string) => {
  // 마스터: 모든 프로그램
  if (user.adminLevel === 'MASTER') return true

  // 담당자: 본인 프로그램
  const programRole = user.programRoles?.[programId]
  return programRole === 'OWNER'
}
```

#### 2) 강사 조회 페이지 (`/admin/users/instructors`)
```typescript
// FR-F00: 강사 조회 (필라별)
interface InstructorListFilters {
  pillar?: string               // 필라(분야) 필터
  status?: 'ACTIVE' | 'INACTIVE'
  keyword?: string
}
```

#### 3) 다운로드 이력 기록 (Mock)
```typescript
// FR-F00: 조회/다운로드 이력 기록

  targetType: 'PARTICIPANTS' | 'INSTRUCTORS'

```

### Hook
- `useParticipantList(filters)` : 참여자 목록 조회
- `useInstructorList(filters)` : 강사 목록 조회
- `useDownloadPermission(programId)` : 다운로드 권한 체크
- `useLogDownload()` : 다운로드 이력 기록

### 화면

- `/admin/users/instructors` : 강사 조회

#### 1) 상태 변경 UI
```typescript
// 상태 변경 드롭다운
const StatusChangeDropdown = ({ applicationId, currentStatus }: Props) => {
  const { changeStatus } = useStatusChange()

  const nextStatuses = getNextStatuses(currentStatus)

  return (
    <Select
      value={currentStatus}
      onChange={(newStatus) => changeStatus(applicationId, newStatus)}
    >
      {nextStatuses.map(status => (
        <Option key={status} value={status}>
          {APPLICATION_STATUS[status].label}
        </Option>
      ))}
    </Select>
  )
}

// 상태 전이 규칙
const getNextStatuses = (current: ApplicationStatus) => {
  const transitions = {
    RECEIVED: ['MATCHING_IN_PROGRESS'],
    MATCHING_IN_PROGRESS: ['MATCHING_COMPLETED'],
    MATCHING_COMPLETED: ['MATERIAL_PREPARING'],
    MATERIAL_PREPARING: ['MATERIAL_SHIPPED'],
    MATERIAL_SHIPPED: ['IN_PROGRESS'],
    IN_PROGRESS: ['SURVEY_SUBMITTED'],
    SURVEY_SUBMITTED: ['REPORT_SUBMITTED'],
    REPORT_SUBMITTED: [],
  }
  return transitions[current] || []
}
```
